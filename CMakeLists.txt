cmake_minimum_required(VERSION 3.16)
project(ScintillatorSimulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Geant4: prefer the build with UI + visualisation, fall back to a plain build.
find_package(Geant4 QUIET COMPONENTS ui_all vis_all)
if(Geant4_FOUND)
  set(HAVE_G4VIS ON)
else()
  find_package(Geant4 REQUIRED)
  set(HAVE_G4VIS OFF)
endif()
find_package(yaml-cpp REQUIRED)

add_executable(CsPbBr3Sim CsPbBr3Sim.cc)
target_include_directories(CsPbBr3Sim PRIVATE ${Geant4_INCLUDE_DIRS})
if(TARGET yaml-cpp::yaml-cpp)
  set(YAML_LIB yaml-cpp::yaml-cpp)
elseif(TARGET yaml-cpp)
  set(YAML_LIB yaml-cpp)
else()
  set(YAML_LIB ${YAML_CPP_LIBRARIES})
endif()
target_link_libraries(CsPbBr3Sim PRIVATE ${Geant4_LIBRARIES} ${YAML_LIB})
if(HAVE_G4VIS)
  target_compile_definitions(CsPbBr3Sim PRIVATE HAVE_G4VIS)
endif()
# Geant4's own flags (-std, -pthread, ...) are appended through the use-file
target_compile_options(CsPbBr3Sim PRIVATE -Wall)
